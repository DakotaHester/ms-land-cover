\section{Introduction}
%------------------------------------------------
\begin{frame}{Deep Learning for Accurate VHSR Land Cover Mapping}
    \begin{itemize}
        \item<1-> DL models incorporate spatial + spectral features to produce detailed maps in an end-to-end manner \citep{yuanReviewDeepLearning2021}.
        \item<2-> VHSR (\(<\)\SI{5}{\meter}) land use/land cover products reveal fine-scale features critical for many applications \citep{liImpactsSpatialResolutions2022}.
        \item<3-> Modern CNN/ViT models are more than capable of crossing the ``resolution barrier'' that plagued earlier methods \citep{woodcockFactorScaleRemote1987, sherrahFullyConvolutionalNetworks2016}.
    \end{itemize}
\end{frame}


\begin{frame}{Practical Barriers to VHSR Land Cover Mapping}
    \vspace{1cm}
    % \textbf{The Problem:}
    \begin{itemize}
        \item<1->DL models are powerful but data-hungry \citep{zhuDeepLearningRemote2017a}.
        \item<2-> Most research into VHSR land cover mapping use benchmark datasets with abundant labels over small areas \citep{2DSemanticLabel,2DSemanticLabeling}
        \item<3-> Creating large, labeled datasets for new areas is a major bottleneck in terms of human capital \citep{huTransferringDeepConvolutional2015}.
    \end{itemize}
\end{frame}

\begin{frame}{A Two-Part Approach to Label-Efficient VHSR Land Cover Mapping}
    \begin{center}
        How can we train accurate models with limited labeled data?
    \end{center}
    \vspace{0.5cm}
    \begin{itemize}
        \item<2-> \textbf{Part 1:} Sample smarter. Use existing, coarse land cover data to guide selection of a small but informative training set based on approximate composition of candidate patches.
        \item<3-> \textbf{Part 2:} Pre-train better. Use self-supervised learning on large amounts VHSR imagery to learn robust, domain-specific features without labels.
    \end{itemize}
\end{frame}

\section{Self-Supervised Learning}

\begin{frame}{Part 2: Better Pre-training with Self-Supervision}
    \begin{itemize}
        \item<1-> \textbf{Idea:} Leverage vast amounts of \textit{unlabeled} imagery to learn meaningful features before fine-tuning on our small labeled dataset.
        \item<2-> \textbf{Method:} Use Bootstrap Your Own Latent (BYOL), a self-supervised learning technique \citep{grillBootstrapYourOwn2020}.

\begin{frame}{Pre-Training Procedure}
    \begin{itemize}
        \item<1-> Pre-train a ResNet-101 backbone with BYOL on a large set of unlabeled NAIP imagery.
        \item<2-> Evaluate learned features using linear probing on a small labeled set.
        \item<3-> Fine-tune full segmentation models (e.g., U-Net, DeepLabv3+) on labeled sets of varying sizes (\(N=250, 500, 750\)).
        \item<4-> Compare performance against standard ImageNet pre-training.
    \end{itemize}
\end{frame}

\section{Mississippi Land Cover Inference}

\begin{frame}{Mississippi Land Cover Product}

        \begin{itemize}
            \item<2-> ImageNet + BYOL pre-training
            \item<3-> U-Net~\citep{ronnebergerUNetConvolutionalNetworks2015}

        \item<6-> Sliding window inference with test-time augmentations \citep{moshkovTesttimeAugmentationDeep2020} to generate land cover map for entire state of Mississippi from 2023 NAIP imagery.
        \item<7-> Use 25,000 randomly sampled, spatially independent points for validation and accuracy assessment
    \end{itemize}
\end{frame}

\section{Results \& Conclusion}
%------------------------------------------------

\begin{frame}{Conclusion}
    \begin{itemize}
        \item<1-> High-quality, large-area VHSR land cover mapping is possible with very limited labeled data.
        % \vspace{1cm}
        \item<2-> \textbf{Key Takeaways:}
        \begin{enumerate}
            \item<3-> \textbf{Sample Smarter:} Semantic stratification using existing coarse data to guide sample selection can be used to build more informative training datasets.
            \item<4-> \textbf{Pre-train Better:} Self-supervised learning on unlabeled domain-specific imagery is highly effective at bridging the domain gap between ImageNet and remote sensing imagery.
        \end{enumerate}
        % \vspace{1cm}
        \item<5-> This work provides a blueprint for label-efficient deep learning in remote sensing.
    \end{itemize}
\end{frame}
