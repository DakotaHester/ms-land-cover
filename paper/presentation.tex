\section{Semantic Stratification}
%------------------------------------------------

\begin{frame}{Part 1: Smarter Sampling with Semantic Stratification}
    \begin{itemize}
        \item<2-> \textbf{Idea:} Don't just sample randomly. Select patches that are representative of the diverse land cover compositions in the study area.
        \item<3-> \textbf{Method:} Use existing, coarse land cover data (like NLCD) to guide patch sampling.
    \end{itemize}

    }

\begin{frame}{Part 2: Better Pre-training with Self-Supervision}
    \begin{itemize}
        \item<1-> \textbf{Idea:} Leverage vast amounts of \alert{unlabeled imagery} (377k) to learn meaningful features before fine-tuning on our small labeled dataset (250, 500, 750).
        \item<2-> \textbf{Method:} Use Bootstrap Your Own Latent (BYOL), a self-supervised learning technique \citep{grillBootstrapYourOwn2020}.

\section{Results \& Conclusion}
%------------------------------------------------

\begin{frame}{Conclusion}
    \begin{itemize}
        \item<1-> High-quality, large-area VHSR land cover mapping is possible with \alert{limited labeled data}.
        % \vspace{1cm}
        \item<2-> \textbf{Key Takeaways:}
        \begin{enumerate}
            \item<3-> \textbf{Sample Smarter:} Novel framework for \alert{patch-based sampling} using existing coarse resolution data leads to more informative training datasets.
            % Semantic stratification using \alert{existing coarse data} to guide sample selection can be used to build more informative training datasets.
            \item<4-> \textbf{Pre-train Better:} \alert{Self-supervised learning} on unlabeled domain-specific imagery is highly effective at bridging the domain gap between ImageNet and remote sensing imagery.
        \end{enumerate}
        % \vspace{1cm}
        % \item<5-> This work provides a blueprint for label-efficient deep learning in remote sensing.
    \end{itemize}
\end{frame}
