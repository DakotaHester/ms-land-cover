\section{Introduction}
%------------------------------------------------
\begin{frame}{Deep learning semantic segmentation models are capable of producing accurate land cover maps at VHSR}
    \begin{itemize}
        \item<1-> DL models incorporate spatial + spectral features to produce detailed maps in an end-to-end manner \citep{}.
        \item<2-> VHSR (\(<\)\SI{5}{\meter}) land use/land cover products reveal fine-scale features critical for many applications \citep{}.
        \item<3-> Modern CNN/ViT models are more than capable of crossing the ``resolution barrier'' that plauged earlier methods \citep{}.
    \end{itemize}
\end{frame}


\begin{frame}{Practical barriers land cover mapping with semantic segmentation at VHSR}
    \vspace{1cm}
    % \textbf{The Problem:}
    \begin{itemize}
        \item DL models are powerful but data-hungry \citep{zhuDeepLearningRemote2017a}.
        \item Most research into VHSR land cover mapping use benchmark datasets with abundant labels over small areas \citep{2DSemanticLabel,2DSemanticLabeling}
        \item Creating large, labeled datasets for new areas is a major bottleneck in terms of human capital
    \end{itemize}
\end{frame}

\begin{frame}{Two-part approach to operational VHSR land cover mapping with few labels}
    \begin{center}
        How can we train accurate models with limited labeled data?
    \end{center}
    \vspace{0.5cm}
    \begin{itemize}
        \item \textbf{Part 1:} Sample smarter. Use existing, coarse land cover data to guide selection of a small but informative training set based on approximate composition of candidate patches.
        \item \textbf{Part 2:} Pre-train better. Use self-supervised learning on large amounts VHSR imagery to learn robust, domain-specific features without labels.
    \end{itemize}
\end{frame}

%------------------------------------------------
\section{Semantic Stratification}
%------------------------------------------------

\begin{frame}{Part 1: Smarter Sampling with Semantic Stratification}
    \begin{itemize}
        \item \textbf{Idea:} Don't just sample randomly. Select patches that are representative of the diverse land cover compositions in the study area.
        \item \textbf{Method:} Use existing, coarse land cover data (like NLCD) to guide our sampling.
    \end{itemize}
    \begin{figure}
        \includegraphics[width=\textwidth]{images/semantic_stratification.drawio.png}
        \caption{We cluster candidate patches based on their land cover mix (from NLCD) and sample from these clusters.}
    \end{figure}
\end{frame}

\begin{frame}{Semantic Stratification: The Result}
    \begin{figure}
        \includegraphics[width=0.9\linewidth]{images/VA_PCs_Layout.png}
        \caption{Patches within the same cluster tend share land cover characteristics.}
    \end{figure}
\end{frame}

\begin{frame}{Validation: Does it Work?}
    \begin{center}
        Yes! Semantic stratification leads to better models.
    \end{center}
    \begin{figure}
        \includegraphics[width=\textwidth]{images/cpb_sampling_methods_boxplot.pdf}
        \caption{Our proposed method shows statistically significant performance gains over random sampling and spatially stratified sampling at \(N_\text{train} = 500\) and \(N_\text{train} = 750\).}
    \end{figure}
\end{frame}

%------------------------------------------------
\section{Self-Supervised Learning}
%------------------------------------------------

\begin{frame}{Part 2: Better Pre-training with Self-Supervision}
    \begin{itemize}
        \item \textbf{Idea:} Leverage vast amounts of \textit{unlabeled} imagery to learn meaningful features before fine-tuning on our small labeled dataset.
        \item \textbf{Method:} Use Bootstrap Your Own Latent (BYOL), a self-supervised learning technique \citep{grillBootstrapYourOwn2020}.
    \end{itemize}
    \begin{figure}
        \includegraphics[width=\textwidth]{images/BYOL_2.drawio.pdf}
        \caption{BYOL trains a network to predict a different augmented view of the same image, learning robust, transformation-invariant features.}
    \end{figure}
\end{frame}

\begin{frame}{Validation: Are the Features Better?}
    \begin{center}
        Yes! The learned features are far more effective for our task.
    \end{center}
    \begin{figure}
        \includegraphics[width=0.7\textwidth]{images/linear_probing_results.pdf}
        \caption{Linear probing shows features from BYOL models vastly outperform standard ImageNet features, closing the domain gap.}
    \end{figure}
\end{frame}

\begin{frame}{Impact on Fine-Tuning Performance}
    \begin{center}
        This translates to better performance on the final segmentation task.
    \end{center}
    \begin{figure}
        \includegraphics[width=\textwidth]{images/finetuning_results.pdf}
        \caption{Fine-tuning various segmentation models shows that BYOL pre-training consistently improves mIoU, especially with few labeled samples.}
    \end{figure}
\end{frame}

\section{Results \& Conclusion}
%------------------------------------------------

\begin{frame}{Final Result: A 1m Land Cover Map of Mississippi}
    \begin{itemize}
        \item Using our two-part strategy, we trained an ensemble of models with only 1,000 labeled samples.
        \item This produced the first publicly-available, 1-meter resolution land cover map for the entire state.
    \end{itemize}
    \begin{figure}
        \includegraphics[width=\textwidth]{images/Inference_Results.png}
        \caption{Our result (left) provides far greater detail than existing products like NLCD (right).}
    \end{figure}
\end{frame}

\begin{frame}{Visual Assessment}
    \begin{figure}
        \includegraphics[width=\textwidth]{images/qualitative_results.png}
        \caption{The model accurately captures fine-scale features like individual buildings, roads, and vegetation boundaries.}
    \end{figure}
\end{frame}

\begin{frame}{Conclusion}
    \begin{itemize}
        \item High-quality, large-area VHSR land cover mapping is possible with very limited labeled data.
        \vspace{1cm}
        \item \textbf{Key Takeaways:}
        \begin{enumerate}
            \item \textbf{Sample Smarter:} Semantic stratification using existing coarse data is a powerful way to build more informative training sets.
            \item \textbf{Pre-train Better:} Self-supervised learning on unlabeled domain-specific imagery is highly effective at bridging the domain gap.
        \end{enumerate}
        \vspace{1cm}
        \item This work provides a blueprint for label-efficient deep learning in remote sensing.
    \end{itemize}
\end{frame}
